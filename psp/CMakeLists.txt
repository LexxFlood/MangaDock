cmake_minimum_required(VERSION 3.16)
project(MangaDock LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(MbedTLS REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf libzip)
find_path(STB_INCLUDE_DIR stb_image.h PATH_SUFFIXES stb REQUIRED)
add_executable(MangaDock main.cpp)
target_include_directories(MangaDock PRIVATE ${STB_INCLUDE_DIR})
target_compile_options(MangaDock PRIVATE -Wall -Wextra)
# Use static transitive dependencies from the SDK .pc files.
target_include_directories(MangaDock PRIVATE ${DEPS_INCLUDE_DIRS})
target_link_directories(MangaDock PRIVATE ${DEPS_STATIC_LIBRARY_DIRS})
target_link_libraries(MangaDock PRIVATE -Wl,--start-group ${DEPS_STATIC_LIBRARIES} -Wl,--end-group)
target_link_options(MangaDock PRIVATE ${DEPS_STATIC_LDFLAGS_OTHER})
if(COMMAND create_pbp_file)
  target_link_libraries(MangaDock PRIVATE pspctrl)
  create_pbp_file(TARGET MangaDock ICON_PATH NULL BACKGROUND_PATH NULL PREVIEW_PATH NULL TITLE MangaDock VERSION 00.10)
endif()
